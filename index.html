    // ================= SILNIK MATEMATYCZNY: INFLATIONARY ELO + ZAKŁADY + SEZONY 2V2 =================

    function computeMatchDelta(ev, currentRatingsMap) {

      if (ev.mode === 'SEASON_TRANSFER' || ev.category === 'SEASON_2V2' || ev.mode === 'BET_TRANSFER' || ev.category === 'BET_ELO') {

        return {

          winnerDelta: 0,

          loserDelta: 0,

          eloSwing: 0,

          shield: 0,

          winProb: 1.0,

          isCoop: false,

          isCustomTransfer: true,

          isBetTransfer: ev.mode === 'BET_TRANSFER' || ev.category === 'BET_ELO',

          customDeltas: ev.customDeltas || {}

        };

      }

      const cat = CATEGORIES[ev.category] || CATEGORIES.IRL_SPORT;

      const winners = (ev.winners || []).filter(id => currentRatingsMap[id] !== undefined);

      const losers = (ev.losers || []).filter(id => currentRatingsMap[id] !== undefined);

      const totalParticipants = winners.length + losers.length;

      if (totalParticipants === 0) {

        return { winnerDelta: 0, loserDelta: 0, eloSwing: 0, shield: 0, winProb: 0.5, isCoop: true };

      }

      const synergyMultiplier = cat.isIrl ? (1 + 0.10 * Math.max(0, totalParticipants - 2)) : 1.0;

      const shield = Math.round((cat.baseShield * synergyMultiplier + (Number(ev.aiBonus) || 0)) * 10) / 10;

      if (ev.mode === 'COOP' || losers.length === 0 || winners.length === 0) {

        const coopGain = Math.round(shield * 10) / 10;

        return {

          winnerDelta: coopGain,

          loserDelta: coopGain,

          eloSwing: 0,

          shield: coopGain,

          winProb: 1.0,

          isCoop: true

        };

      }

      const avgWinElo = winners.reduce((s, id) => s + currentRatingsMap[id], 0) / winners.length;

      const avgLoseElo = losers.reduce((s, id) => s + currentRatingsMap[id], 0) / losers.length;

      const expectedWin = 1 / (1 + Math.pow(10, (avgLoseElo - avgWinElo) / 400));

      const eloSwing = Math.round(cat.kFactor * (1 - expectedWin) * 10) / 10;

      const winnerDelta = Math.round((eloSwing + shield) * 10) / 10;

      const loserDelta = Math.round((-eloSwing + shield) * 10) / 10;

      return {

        winnerDelta,

        loserDelta,

        eloSwing,

        shield,

        winProb: expectedWin,

        avgWinElo: Math.round(avgWinElo),

        avgLoseElo: Math.round(avgLoseElo),

        isCoop: false

      };

    }

    function runChronologicalSimulation() {

      const statsMap = {};

      const ratingsMap = {};

      appState.members.forEach(m => {

        ratingsMap[m.id] = START_ELO;

        statsMap[m.id] = {

          member: m,

          elo: START_ELO,

          wins: 0,

          losses: 0,

          coops: 0,

          skillEloNet: 0,

          shieldEloNet: 0,

          season2v2Net: 0,

          betEloNet: 0,

          irlCount: 0,

          onlineCount: 0,

          totalMinutes: 0,

          form: []

        };

      });

      const chronologicalEvents = [...appState.events].sort((a, b) => {

        if (a.date !== b.date) return a.date.localeCompare(b.date);

        return String(a.id).localeCompare(String(b.id));

      });

      const eventComputedMap = {};

      chronologicalEvents.forEach(ev => {

        const calc = computeMatchDelta(ev, ratingsMap);

        eventComputedMap[ev.id] = calc;

        const cat = CATEGORIES[ev.category] || CATEGORIES.IRL_SPORT;

        // Obsługa transferu z Zakładu ELO lub Sezonu Ligi 2v2

        if (calc.isCustomTransfer && ev.customDeltas) {

          Object.entries(ev.customDeltas).forEach(([pid, rawDelta]) => {

            const delta = Math.round((Number(rawDelta) || 0) * 10) / 10;

            if (statsMap[pid] !== undefined) {

              ratingsMap[pid] = Math.round((ratingsMap[pid] + delta) * 10) / 10;

              const st = statsMap[pid];

              st.elo = ratingsMap[pid];

              if (calc.isBetTransfer) {

                st.betEloNet = Math.round((st.betEloNet + delta) * 10) / 10;

              } else {

                st.season2v2Net = Math.round((st.season2v2Net + delta) * 10) / 10;

                st.onlineCount += 1;

              }

              if (delta > 0) st.form.push('W');

              else if (delta < 0) st.form.push('L');

              else st.form.push('C');

            }

          });

          return;

        }

        const winners = (ev.winners || []).filter(id => statsMap[id]);

        const losers = (ev.losers || []).filter(id => statsMap[id]);

        if (calc.isCoop) {

          [...winners, ...losers].forEach(pid => {

            ratingsMap[pid] = Math.round((ratingsMap[pid] + calc.winnerDelta) * 10) / 10;

            const st = statsMap[pid];

            st.elo = ratingsMap[pid];

            st.coops += 1;

            st.shieldEloNet += calc.shield;

            if (cat.isIrl) st.irlCount += 1; else st.onlineCount += 1;

            st.totalMinutes += Number(ev.duration) || 0;

            st.form.push('C');

          });

        } else {

          winners.forEach(pid => {

            ratingsMap[pid] = Math.round((ratingsMap[pid] + calc.winnerDelta) * 10) / 10;

            const st = statsMap[pid];

            st.elo = ratingsMap[pid];

            st.wins += 1;

            st.skillEloNet += calc.eloSwing;

            st.shieldEloNet += calc.shield;

            if (cat.isIrl) st.irlCount += 1; else st.onlineCount += 1;

            st.totalMinutes += Number(ev.duration) || 0;

            st.form.push('W');

          });

          losers.forEach(pid => {

            ratingsMap[pid] = Math.round((ratingsMap[pid] + calc.loserDelta) * 10) / 10;

            const st = statsMap[pid];

            st.elo = ratingsMap[pid];

            st.losses += 1;

            st.skillEloNet -= calc.eloSwing;

            st.shieldEloNet += calc.shield;

            if (cat.isIrl) st.irlCount += 1; else st.onlineCount += 1;

            st.totalMinutes += Number(ev.duration) || 0;

            st.form.push('L');

          });

        }

      });

      const sortedLeaderboard = Object.values(statsMap).sort((a, b) => b.elo - a.elo || b.wins - a.wins);

      return { sortedLeaderboard, ratingsMap, eventComputedMap };

    }

    function computeBetResolutionDeltas(bet, winnersSet) {

      const participants = bet.participants || [];

      const winners = participants.filter(id => winnersSet.has(id));

      const losers = participants.filter(id => !winnersSet.has(id));

      const stake = Number(bet.stake) || 20;

      if (winners.length === 0 || losers.length === 0) return null;

      // Każdy przegrany traci dokładnie swoją stawkę (-stake), a suma ich strat dzielona jest równo między zwycięzców

      const totalLostPot = losers.length * stake;

      const winGainPerWinner = Math.round((totalLostPot / winners.length) * 10) / 10;

      const customDeltas = {};

      winners.forEach(id => { customDeltas[id] = winGainPerWinner; });

      losers.forEach(id => { customDeltas[id] = -stake; });

      return { winners, losers, customDeltas, winGainPerWinner, stake };

    }

    async function resolveWorkingModel(apiKey) {

      const preferred = appConfig.model || 'gemini-2.5-flash';

      try {

        const res = await fetch(`https://generativelanguage.googleapis.com/v1beta/models?key=${encodeURIComponent(apiKey)}`);

        const data = await res.json();

        if (data.models && Array.isArray(data.models)) {

          const supported = data.models

            .filter(m => (m.supportedGenerationMethods || []).includes('generateContent'))

            .map(m => m.name.replace(/^models\//, ''));

          if (supported.includes(preferred)) return [preferred, ...supported.filter(x => x !== preferred)];

          const prioritized = [

            ...FALLBACK_MODELS.filter(m => supported.includes(m)),

            ...supported.filter(m => m.includes('flash'))

          ];

          if (prioritized.length > 0) return [...new Set(prioritized)];

        }

      } catch (e) {}

      return [...new Set([preferred, ...FALLBACK_MODELS])];

    }

    async function callGeminiWithFallback(apiKey, payload, statusElId = null) {

      const candidates = await resolveWorkingModel(apiKey);

      let lastError = null;

      for (const modelName of candidates) {

        if (statusElId) {

          const statusEl = document.getElementById(statusElId);

          if (statusEl) statusEl.textContent = `Przetwarzanie przez ${modelName}...`;

        }

        const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/${modelName}:generateContent?key=${encodeURIComponent(apiKey)}`, {

          method: 'POST',

          headers: { 'Content-Type': 'application/json' },

          body: JSON.stringify(payload)

        });

        const data = await response.json();

        if (!data.error) {

          appConfig.model = modelName;

          localStorage.setItem(STORAGE_KEY_CFG, JSON.stringify(appConfig));

          updateConfigStatuses();

          return data;

        }

        lastError = data.error.message || JSON.stringify(data.error);

      }

      throw new Error(lastError || 'Błąd połączenia z API Gemini.');

    }

    function findMembersInTextSegment(segment) {

      const segLow = (segment || '').toLowerCase();

      const found = [];

      appState.members.forEach(m => {

        const stem = m.name.toLowerCase().slice(0, Math.max(3, m.name.length - 1));

        const aliasLow = (m.alias2v2 || '').toLowerCase();

        if (segLow.includes(stem) || (aliasLow && segLow.includes(aliasLow))) found.push(m.id);

      });

      return found;

    }

    function runLocalEloHeuristicJudge(text, manualCtx) {

      const lower = text.toLowerCase();

      let category = 'IRL_SPORT';

      if (/(lol|2v2|cs|valorant|discord|online|komp|custom)/.test(lower)) category = 'ONLINE_GAMING';

      else if (/(wyjazd|góry|mazury|weekend|wypad|namiot)/.test(lower)) category = 'IRL_TRIP';

      else if (/(kręgl|kregl|bilard|dart|rzutk|grill|piwo|pizza|plansz|kino|chill|ognisko|bar)/.test(lower)) category = 'IRL_CHILL';

      let duration = category === 'ONLINE_GAMING' ? 60 : 120;

      const hourMatch = lower.match(/(\d+(?:[.,]\d+)?)\s*(h|godz)/);

      if (hourMatch) duration = Math.round(parseFloat(hourMatch[1].replace(',', '.')) * 60);

      let aiBonus = 0;

      if (/(deszcz|ulew|śnieg|mróz)/.test(lower)) aiBonus += 4;

      if (/(transport|ogarnął|kierowc)/.test(lower)) aiBonus += 3;

      const mentionedInText = findMembersInTextSegment(text);

      // Jeśli użytkownik zaznaczył wcześniej osoby A, B, C i w opisie ("byliśmy na kręglach") nie wpisuje nowych imion:

      if (manualCtx.hasManualSelection && mentionedInText.length === 0) {

        const hasBothSides = manualCtx.preWinners.length > 0 && manualCtx.preLosers.length > 0;

        const mode = hasBothSides ? 'VERSUS' : (manualCtx.preCoop.length > 0 ? 'COOP' : document.getElementById('evMode').value);

        const cleanActivity = text.charAt(0).toUpperCase() + text.slice(1);

        const title = hasBothSides

          ? `${cleanActivity} (Wygrana: ${manualCtx.preWinners.join(', ')})`

          : cleanActivity;

        const comment = hasBothSides

          ? `Sędzia AI: ${manualCtx.preWinners.join(', ')} wygrywa rywalizację (${cleanActivity}) przeciwko: ${manualCtx.preLosers.join(', ')}!`

          : `Sędzia AI: Wspólne wyjście ekipy (${cleanActivity}) – wszyscy obecni zgarniają Tarczę Aktywności!`;

        applyJudgeToForm({

          title,

          category,

          mode,

          duration,

          aiBonus,

          keepExistingParticipants: true,

          comment

        }, manualCtx, text);

        return;

      }

      // Standardowe wykrywanie z tekstu gdy wpisano imiona w tekście

      let mode = 'VERSUS';

      let winnersIds = [];

      let losersIds = [];

      if (/(wygra|pokona|ogra|przegra|vs|\d+:\d+)/i.test(lower)) {

        const vsSplit = text.split(/ (?:z|ze|vs|przeciwko) /i);

        if (vsSplit.length >= 2) {

          winnersIds = findMembersInTextSegment(vsSplit[0]);

          losersIds = findMembersInTextSegment(vsSplit.slice(1).join(' ')).filter(id => !winnersIds.includes(id));

        }

      }

      if (winnersIds.length === 0 || losersIds.length === 0) {

        if (category === 'IRL_CHILL' || category === 'IRL_TRIP' || mentionedInText.length < 2) {

          mode = 'COOP';

          winnersIds = mentionedInText;

          losersIds = [];

        } else {

          const half = Math.ceil(mentionedInText.length / 2);

          winnersIds = mentionedInText.slice(0, half);

          losersIds = mentionedInText.slice(half);

        }

      }

      applyJudgeToForm({

        title: text.length > 55 ? text.slice(0, 55).trim() + '...' : (text.charAt(0).toUpperCase() + text.slice(1)),

        category,

        mode,

        duration,

        aiBonus,

        winnersIds,

        losersIds,

        comment: `Sędzia AI: Rozpoznano ${mode === 'VERSUS' ? 'pojedynek ELO' : 'kooperację'} (${CATEGORIES[category].label}).`

      }, manualCtx, text);

    }

    function applyJudgeToForm(res, manualCtx, rawText = '') {

      if (res.title) document.getElementById('evTitle').value = res.title;

      if (res.category && CATEGORIES[res.category]) document.getElementById('evCategory').value = res.category;

      if (res.duration) document.getElementById('evDuration').value = res.duration;

      if (typeof res.aiBonus === 'number') document.getElementById('evAiBonus').value = res.aiBonus;

      if (res.comment) document.getElementById('evComment').value = res.comment;

      const mentionedInRawText = findMembersInTextSegment(rawText);

      // Jeśli użytkownik miał już zaznaczone osoby na górze i w opisie nie podał żadnych nowych imion -> NIE NADPISUJ JEGO WYBORU OSÓB!

      if (res.keepExistingParticipants || (manualCtx && manualCtx.hasManualSelection && mentionedInRawText.length === 0)) {

        if (manualCtx.preWinners.length > 0 && manualCtx.preLosers.length > 0) {

          document.getElementById('evMode').value = 'VERSUS';

        } else if (res.mode) {

          document.getElementById('evMode').value = res.mode;

        }

        renderParticipantsSelector();

        return;

      }

      if (res.mode) document.getElementById('evMode').value = res.mode;

      participantRoles = {};

      const matchName = (name) => {

        const clean = String(name).toLowerCase().slice(0, 3);

        return appState.members.find(m =>

          m.name.toLowerCase().includes(clean) ||

          (m.alias2v2 && m.alias2v2.toLowerCase().includes(clean))

        );

      };

      if (Array.isArray(res.winnersIds)) {

        res.winnersIds.forEach(id => { participantRoles[id] = res.mode === 'COOP' ? 'COOP' : 'WIN'; });

      } else if (Array.isArray(res.winnersNames)) {

        res.winnersNames.forEach(n => {

          const m = matchName(n);

          if (m) participantRoles[m.id] = res.mode === 'COOP' ? 'COOP' : 'WIN';

        });

      }

      if (res.mode !== 'COOP') {

        if (Array.isArray(res.losersIds)) {

          res.losersIds.forEach(id => { if (!participantRoles[id]) participantRoles[id] = 'LOSE'; });

        } else if (Array.isArray(res.losersNames)) {

          res.losersNames.forEach(n => {

            const m = matchName(n);

            if (m && !participantRoles[m.id]) participantRoles[m.id] = 'LOSE';

          });

        }

      }

      renderParticipantsSelector();

    }

    function normalizePlayerName2v2(rawName) {

      if (!rawName) return '';

      const trimmed = rawName.trim();

      const lower = trimmed.toLowerCase();

      if (ALIAS_MAP_2V2[lower]) return ALIAS_MAP_2V2[lower];

      const existingKey = Object.keys(appState.league2v2.players).find(k => k.toLowerCase() === lower);

      if (existingKey) return existingKey;

      const byMemberAlias = appState.members.find(m => m.alias2v2 && m.alias2v2.toLowerCase() === lower);

      if (byMemberAlias) return byMemberAlias.alias2v2;

      return trimmed;

    }
